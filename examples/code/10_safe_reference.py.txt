# GÜVENLİ REFERANS — bilerek temiz tutuldu
# Neden "zafiyet yok" sayılmalı?
# - Secret env'den geliyor
# - SQL parametreli
# - subprocess shell=False + argüman listesi
# - TLS verify=True
# - Path kök dizinin dışına çıkamıyor
# - Token için secrets modülü (kriptografik PRNG)

import os
import sqlite3
import subprocess
import secrets
import requests
from pathlib import Path

API_KEY = os.environ["API_KEY"]
BASE = Path("/var/data").resolve()

def get_user(conn: sqlite3.Connection, user_id: int):
    return conn.execute(
        "SELECT id, name FROM users WHERE id = ?",
        (user_id,),
    ).fetchone()

def run_safe(args: list[str]):
    subprocess.run(args, shell=False, check=True)

def fetch(url: str):
    return requests.get(url, timeout=10, verify=True)

def read_user_file(name: str) -> str:
    target = (BASE / Path(name).name).resolve()
    if not str(target).startswith(str(BASE)):
        raise ValueError("path traversal blocked")
    return target.read_text(encoding="utf-8")

def new_token() -> str:
    return secrets.token_urlsafe(32)
