# DOĞRUSU: Path(name).name ile sadece dosya adı al; resolve() ile kök dizini doğrula.

from flask import send_file

def read_user_file(name):
    return open("/var/data/" + name).read()

def download(name):
    return send_file("../uploads/" + name)
